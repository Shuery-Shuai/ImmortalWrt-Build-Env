sudo
bash

# --- bootstrap（刻意例外：脚本自身在安装任何东西之前就要用它们）---
# 脚本在第 104 行 check_network 起就调用 curl，并在 update_apt_source 中用它拉取第三方源与
# GPG key；此时脚本自己的 apt 安装尚未开始。ca-certificates 是这些 HTTPS 请求的前提，
# 不能依赖 curl 的 Recommends（--no-install-recommends 会移除它）。
curl
ca-certificates

# --- 脚本不安装、但本镜像必须有的包 ---
# openssh-client：git 的 Recommends，缺了它 `git clone git@…` 与 ssh 子模块会失败（CI 实测发现）。
openssh-client
# gcc-multilib / g++-multilib：与脚本安装的 gcc-12-multilib / g++-12-multilib 等效，
#   刻意保留，以便脚本安装路径变化时仍有 multilib 工具链。
gcc-multilib
g++-multilib
# libgmp-dev / libgnutls28-dev：脚本用的是过渡名 libgmp3-dev / gnutls-dev（由这两个包 Provides），
#   这里用正式名显式声明。
libgmp-dev
libgnutls28-dev
# qemu-block-extra：qemu-img 的额外块驱动（vmdk/vdi 等）；x86 目标生成 VMware/VirtualBox 镜像时用到。
qemu-block-extra
